Route & Controller
Route::resource('posts', PostController::class);

public function index(Request $r) {
    $posts = Post::when($r->q, fn($q) =>
            $q->where('title', 'like', "%{$r->q}%"))
        ->latest()->paginate(4)->withQueryString();
    return view('posts.index', compact('posts'));
}
public function store(StorePostRequest $r) {
    Post::create($r->validated());
    return to_route('posts.index')->with('success', 'Resep dibuat!');
}
public function update(StorePostRequest $r, Post $post) {
    $post->update($r->validated());
    return to_route('posts.index')->with('success', 'Resep diperbarui!');
}
public function destroy(Post $post) {
    $post->delete();
    return to_route('posts.index')->with('success', 'Resep dihapus!');
}
